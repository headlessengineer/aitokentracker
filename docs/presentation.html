SQLite cache -- mtime invalidation

// src/lib/cache.ts
// schema: plugin_cache(pluginId, days, collectedAt, data)
// path: ~/.config/aitokentracker/cache.db

async function withCache(pluginId, days, dataPath, fn) {
  const maxMtime = maxMtimeMs(dataPath)
  const cached  = getCache(pluginId, days)

  if (cached && maxMtime <= cached.collectedAt) {
    return cached.data  // cache hit: <5ms
  }

  const fresh = await fn()
  setCache(pluginId, days, fresh)
  return fresh
}
// dataPath:'' (Cursor) bypasses mtime check

Open Architecture

Add a new tool in an afternoon

// 1. Create src/plugins/mytool/index.ts
const MY_PLUGIN: TokenPlugin = {
  id: 'mytool', name: 'My Tool', icon: 'MT',
  dataPath: path.join(HOME, '.mytool/sessions'),
  capabilities: { cost: true, models: true, sessions: true },

  async isAvailable() {
    return availResult(await pathExists(DATA_DIR))
    // true  -> { available: true }
    // false -> { available: false, reason: 'path_missing' }
  },

  async collect(options?) {
    const conversations: ConversationSummary[] = /* parse */ []
    if (!conversations.length) return emptyPluginData('mytool')
    return buildPluginData('mytool', conversations, options)
    // handles: daily agg, cost, models, projects, hourly heatmap
  }
}

// 2. Register in src/plugins/index.ts
registry.register(MY_PLUGIN)
